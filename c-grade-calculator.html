#include <stdio.h>

int main() {
    float marks;
    char grade;
    
    printf("Enter your marks (0-100): ");
    scanf("%f", &marks);
    
    if (marks >= 90) {
        grade = 'A';
        printf("Grade: %c - Outstanding Performance!\\n", grade);
    }
    else if (marks >= 80) {
        grade = 'B';
        printf("Grade: %c - Excellent Work!\\n", grade);
    }
    else if (marks >= 70) {
        grade = 'C';
        printf("Grade: %c - Very Good!\\n", grade);
    }
    else if (marks >= 60) {
        grade = 'D';
        printf("Grade: %c - Good Job!\\n", grade);
    }
    else if (marks >= 50) {
        grade = 'E';
        printf("Grade: %c - Average Performance\\n", grade);
    }
    else {
        grade = 'F';
        printf("Grade: %c - Failed - Keep Trying!\\n", grade);
    }
    
    return 0;
}
