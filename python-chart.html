import matplotlib.pyplot as plt

# Data for the chart
labels = []
values = []

print("=== Bar Chart Visualization ===")
print("Enter data points (type 'done' to finish)")

while True:
    label = input("Enter label (or 'done' to finish): ")
    if label.lower() == 'done':
        break
    
    try:
        value = float(input("Enter value: "))
        labels.append(label)
        values.append(value)
        print(f"Added: {label} = {value}")
    except ValueError:
        print("Invalid value! Please enter a number.")

if len(labels) > 0:
    # Create bar chart
    plt.figure(figsize=(10, 6))
    plt.bar(labels, values, color=['#667eea', '#764ba2', '#f093fb', 
                                     '#f5576c', '#06beb6'])
    
    plt.xlabel('Labels', fontsize=12)
    plt.ylabel('Values', fontsize=12)
    plt.title('Bar Chart Visualization', fontsize=14, fontweight='bold')
    plt.xticks(rotation=45, ha='right')
    plt.tight_layout()
    
    # Display the chart
    plt.show()
    
    # Print summary
    print("\\n=== Chart Summary ===")
    for i in range(len(labels)):
        print(f"{labels[i]}: {values[i]}")
else:
    print("No data entered!")
